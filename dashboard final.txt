      const toastContainer = document.getElementById('toastContainer');

      // ─── Toast ───
      function showToast(message, type = 'info') {
        const toast = document.createElement('div');
        toast.className = `toast ${type}`;
        const icon = type === 'success' ? '\u2713' : type === 'error' ? '\u26A0' : '\u2139';
        toast.innerHTML = `<span>${icon}</span><span>${message}</span>`;
        toastContainer.appendChild(toast);
        setTimeout(() => {
          toast.classList.add('removing');
          setTimeout(() => toast.remove(), 300);
        }, 3500);
      }

      // ─── Helpers ───
      function formatDate(dateStr) {
        if (!dateStr) return '\u2014';
        const d = new Date(dateStr);
        if (isNaN(d)) return dateStr;
        return d.toLocaleDateString('en-US', {
          year: 'numeric',
          month: 'short',
          day: 'numeric',
        });
      }

      function statusBadgeClass(status) {
        const map = {
          'Not Started': 'badge-not-started',
          'In Progress': 'badge-in-progress',
          'Completed': 'badge-completed',
        };
        return map[status] || 'badge-not-started';
      }

      function escapeHtml(str) {
        if (str == null) return '';
        const div = document.createElement('div');
        div.textContent = String(str);
        return div.innerHTML;
      }

      function showButtonLoading(btn, textSpan) {
        const originalText = textSpan.textContent;
        textSpan.textContent = '';
        const spinner = document.createElement('span');
        spinner.className = 'spinner';
        btn.appendChild(spinner);
        btn.disabled = true;
        return () => {
          spinner.remove();
          textSpan.textContent = originalText;
          btn.disabled = false;
        };
      }

      // ─── API ───
      async function apiRequest(method, path, body) {
        const options = {
          method,
          headers: { 'Content-Type': 'application/json' },
        };
        if (body) options.body = JSON.stringify(body);
        const res = await fetch(`${API_BASE}${path}`, options);
        if (!res.ok) {
          let errMsg = `Request failed (${res.status})`;
          try {
            const errData = await res.json();
            errMsg = errData.message || errData.error || errMsg;
          } catch (_) {}
          throw new Error(errMsg);
        }
        if (res.status === 204) return null;
            return res.json();
      }

      // ─── Render ───
      function renderStats() {
        const total = courses.length;
        const completed = courses.filter(c => c.status === 'Completed').length;
        const inProgress = courses.filter(c => c.status === 'In Progress').length;
        const notStarted = courses.filter(c => c.status === 'Not Started').length;
        document.getElementById('statTotal').textContent = total;
        document.getElementById('statCompleted').textContent = completed;
        document.getElementById('statProgress').textContent = inProgress;
        document.getElementById('statNotStarted').textContent = notStarted;
      }

      function renderTable() {
        if (isLoading) {
          courseTableContainer.innerHTML = `
            <div class="loading-state">
              <div class="spinner large"></div>
              <span>Loading courses...</span>
            </div>`;
          return;
        }

        if (courses.length === 0) {
          courseTableContainer.innerHTML = `
            <div class="empty-state">
              <div class="empty-icon">&#128218;</div>
              <p>No courses yet. Add your first course above to get started!</p>
            </div>`;
          return;
        }

        courseTableContainer.innerHTML = `
          <div class="table-wrapper">
            <table>
              <thead>
                <tr>
                  <th>Course</th>
                  <th>Target Date</th>
                  <th>Status</th>
                  <th>Created</th>
                  <th>Actions</th>
                </tr>
              </thead>
              <tbody>
                ${courses.map(c => `
                  <tr data-id="${c.id}">
                    <td>
                      <div class="course-name">${escapeHtml(c.name)}</div>
                      <div class="course-desc">${escapeHtml(c.description)}</div>
                    </td>
                    <td>${formatDate(c.target_date)}</td>
                    <td><span class="badge ${statusBadgeClass(c.status)}">${escapeHtml(c.status)}</span></td>
                    <td>${formatDate(c.created_at)}</td>
                    <td>
                      <div class="actions-cell">
                        <button class="btn btn-edit" onclick="openEditModal(${c.id})">Edit</button>
                        <button class="btn btn-delete" onclick="handleDelete(${c.id}, '${escapeHtml(c.name).replace(/'/g, "\\'")}')">Remove</button>
                      </div>
                    </td>
                  </tr>`).join('')}
              </tbody>
            </table>
          </div>`;
      }

      function renderAll() {
        renderTable();
        renderStats();
      }

      // ─── Fetch all ───
      async function fetchCourses() {
        isLoading = true;
        renderTable();
        try {
          const data = await apiRequest('GET', '');
          courses = Array.isArray(data) ? data : ((data && (data.data || data.courses)) || []);
          showToast('Courses loaded successfully', 'success');
        } catch (err) {
          showToast(`Failed to load courses: ${err.message}`, 'error');
          courseTableContainer.innerHTML = `
            <div class="empty-state">
              <div class="empty-icon">\u26A0</div>
              <p>Could not load courses. Make sure the API server is running at ${API_BASE}</p>
            </div>`;
          courses = [];
        } finally {
          isLoading = false;
          renderAll();
        }
      }

      // ─── Add course ───
      addCourseForm.addEventListener('submit', async (e) => {
        e.preventDefault();
        const payload = {
          name: document.getElementById('addName').value.trim(),
          description: document.getElementById('addDescription').value.trim(),
          target_date: document.getElementById('addTargetDate').value,
          status: document.getElementById('addStatus').value,
        };

        if (!payload.name || !payload.description || !payload.target_date) {
          showToast('Please fill in all required fields', 'error');
          return;
        }

        const restoreBtn = showButtonLoading(addSubmitBtn, addBtnText);
        try {
          const newCourse = await apiRequest('POST', '', payload);
          courses.push(newCourse);
          renderAll();
          addCourseForm.reset();
          showToast(`Course "${payload.name}" added successfully`, 'success');
        } catch (err) {
          showToast(`Failed to add course: ${err.message}`, 'error');
        } finally {
          restoreBtn();
        }
      });

      // ─── Edit modal ───
      function openEditModal(id) {
        const course = courses.find(c => c.id === id);
        if (!course) return;
        document.getElementById('editId').value = course.id;
        document.getElementById('editName').value = course.name;
        document.getElementById('editDescription').value = course.description || '';
        document.getElementById('editTargetDate').value = course.target_date || '';
        document.getElementById('editStatus').value = course.status || 'Not Started';
        editModal.classList.add('active');
      }

      function closeEditModal() {
        editModal.classList.remove('active');
      }

      modalCloseBtn.addEventListener('click', closeEditModal);
      modalCancelBtn.addEventListener('click', closeEditModal);
      editModal.addEventListener('click', (e) => {
        if (e.target === editModal) closeEditModal();
      });

      editSubmitBtn.addEventListener('click', async () => {
        const id = document.getElementById('editId').value;
        const payload = {
          name: document.getElementById('editName').value.trim(),
          description: document.getElementById('editDescription').value.trim(),
          target_date: document.getElementById('editTargetDate').value,
          status: document.getElementById('editStatus').value,
        };

        if (!payload.name || !payload.description || !payload.target_date) {
          showToast('Please fill in all required fields', 'error');
          return;
        }

        const restoreBtn = showButtonLoading(editSubmitBtn, editBtnText);
        try {
          const updated = await apiRequest('PUT', `/${id}`, payload);
          const idx = courses.findIndex(c => c.id === parseInt(id) || c.id === id);
          if (idx !== -1) {
            courses[idx] = updated || { ...courses[idx], ...payload };
          }
          renderAll();
          closeEditModal();
          showToast('Course updated successfully', 'success');
        } catch (err) {
          showToast(`Failed to update course: ${err.message}`, 'error');
        } finally {
          restoreBtn();
        }
      });

      // ─── Delete ───
      async function handleDelete(id, name) {
        if (!confirm(`Are you sure you want to remove "${name}"?`)) return;
        const row = document.querySelector(`tr[data-id="${id}"]`);
        if (row) {
          row.style.opacity = '0.5';
          row.style.pointerEvents = 'none';
        }
        try {
          await apiRequest('DELETE', `/${id}`);
          courses = courses.filter(c => c.id !== id && c.id !== parseInt(id));
          renderAll();
          showToast(`Course "${name}" removed`, 'success');
        } catch (err) {
          showToast(`Failed to delete course: ${err.message}`, 'error');
          if (row) {
            row.style.opacity = '1';
            row.style.pointerEvents = 'auto';
          }
        }
      }

      // Expose for inline onclick handlers
      window.openEditModal = openEditModal;
      window.handleDelete = handleDelete;

      // ─── Init ───
      document.addEventListener('DOMContentLoaded', fetchCourses);
    </script>
